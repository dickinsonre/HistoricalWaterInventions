# Adding SWMM5 Export Feature to Historical Mystery

This is a **brilliant** idea that uniquely connects your water history education platform to your professional SWMM5 expertise! Let me design this feature.

## **Concept: "Ancient to Modern" SWMM5 Export**

Users can explore ancient water systems and then export a SWMM5 model that recreates the hydraulics using modern simulation tools. This bridges 8,000 years of history with contemporary engineering practice.

function generateSWMM5File(inventionId, complexity = 'simple') {
  const model = SWMM5_MODELS[inventionId];
  if (!model) {
    alert('SWMM5 model not available for this invention yet.');

  }
  
  const timestamp = new Date().toISOString();
  const p = model.parameters;
  
  let inp = `[TITLE]
;;Project Title/Notes

${model.description}

;; Generated by Historical Mystery - Ancient Water Engineering Explorer
;; Created by Robert Dickinson (50+ years SWMM expertise)
;; Export Date: ${timestamp}

;; Learn more: https://historical-mystery-robertdickinson.replit.app
;;
;; EDUCATIONAL NOTES:
;; - Ancient engineers achieved remarkable precision without modern tools
;; - Many of these systems operated for centuries with minimal maintenance
;; - The hydraulic principles remain valid in modern engineering

[OPTIONS]
;;Option             Value
FLOW_UNITS           CMS
INFILTRATION         HORTON
FLOW_ROUTING         DYNWAVE
LINK_OFFSETS         DEPTH
MIN_SLOPE            0.0001
ALLOW_PONDING        NO
SKIP_STEADY_STATE    NO

START_TIME           00:00:00
REPORT_START_DATE    01/01/2024
REPORT_START_TIME    00:00:00

LAT_FLOW_TOL         5
MINIMUM_STEP         0.5

`;

  // Generate model-specific sections
  switch(inventionId) {
    case 'roman-aqueduct':
      inp += generateAqueductModel(p, complexity);
      break;
    case 'qanat':
      inp += generateQanatModel(p, complexity);
      break;
    case 'egyptian-canal':
      inp += generateCanalModel(p, complexity);
      break;
    case 'dujiangyan':
      inp += generateDujiangyanModel(p, complexity);

    case 'baray':

    case 'minoan-drainage':

    default:

  }
  
  // Add report section
  inp += `

SUBCATCHMENTS ALL
NODES ALL
LINKS ALL

[TAGS]

[MAP]
DIMENSIONS 0.000 0.000 10000.000 10000.000
Units      Meters

[COORDINATES]
;;Node           X-Coord            Y-Coord
`;

  return inp;
}

/**
 * Roman Aqueduct SWMM5 Model
 */
function generateAqueductModel(p, complexity) {
  const numSegments = complexity === 'detailed' ? 10 : 5;
  const segmentLength = p.length / numSegments;
  const elevationDrop = p.length * p.slope;
  
  let model = `
;; ============================================
;; ROMAN AQUEDUCT MODEL
;; ============================================
;; Historical Context:
;; - Romans built 11 major aqueducts serving Rome
;; - Aqua Appia (312 BCE) was the first
;; - Aqua Claudia stretched 69 km
;; - Total capacity: ~1 million m³/day to Rome
;;
;; Engineering Achievements:
;; - Maintained 0.1-0.3% grades over tens of kilometers
;; - Used surveying tool called "groma" and "chorobates"
;; - Combined tunnels, channels, and arcades
;; ============================================

[JUNCTIONS]
;;Name           Elevation      MaxDepth       InitDepth      SurDepth       Aponded
`;

  // Create junction nodes along aqueduct
  for (let i = 0; i <= numSegments; i++) {
    const elevation = 100 - (i * elevationDrop / numSegments);
    const nodeName = i === 0 ? 'SOURCE' : i === numSegments ? 'CASTELLUM' : `NODE_${i}`;
    const comment = i === 0 ? ';; Mountain spring source' : 
                    i === numSegments ? ';; Distribution basin (castellum divisorium)' :
                    `;; Aqueduct segment ${i}`;
    model += `${comment}\n`;
    model += `${nodeName.padEnd(16)} ${elevation.toFixed(2).padStart(10)}     ${(p.channelDepth + 0.5).toFixed(2).padStart(10)}     0              0              0\n`;
  }

  model += `
[OUTFALLS]
;;Name           Elevation      Type       Stage Data       Gated    Route To
;; City distribution point
CITY             ${(100 - elevationDrop - 1).toFixed(2)}          FREE                        NO

[CONDUITS]
;;Name           From Node        To Node          Length     Roughness  InOffset   OutOffset  InitFlow   MaxFlow
;; Aqueduct channel sections - Roman opus signinum (waterproof concrete)
`;

  // Create conduit links
  for (let i = 0; i < numSegments; i++) {
    const fromNode = i === 0 ? 'SOURCE' : `NODE_${i}`;
    const toNode = i === numSegments - 1 ? 'CASTELLUM' : `NODE_${i + 1}`;
    const linkName = `SPECUS_${i + 1}`;  // Roman term for aqueduct channel
    model += `${linkName.padEnd(16)} ${fromNode.padEnd(16)} ${toNode.padEnd(16)} ${segmentLength.toFixed(0).padStart(10)}     ${p.roughness.toFixed(4)}     0          0          0          0\n`;
  }

  // Final link to city
  model += `OUTLET           CASTELLUM        CITY             100        ${p.roughness.toFixed(4)}     0          0          0          0\n`;

  model += `
[XSECTIONS]
;;Link           Shape        Geom1            Geom2      Geom3      Geom4      Barrels    Culvert
;; Rectangular channel (specus) - typical Roman design
`;

  for (let i = 0; i < numSegments; i++) {
    model += `SPECUS_${i + 1}       RECT_OPEN    ${p.channelDepth}            ${p.channelWidth}      0          0          1\n`;
  }
  model += `OUTLET           RECT_OPEN    ${p.channelDepth}            ${p.channelWidth}      0          0          1\n`;

  model += `
[INFLOWS]
;;Node           Constituent      Time Series      Type     Mfactor  Sfactor  Baseline Pattern
;; Spring source - constant flow typical of Roman springs
SOURCE           FLOW             SPRING_FLOW      FLOW     1.0      1.0      ${p.flowRate}

[TIMESERIES]
;;Name           Date       Time       Value
;; Constant spring flow (Roman springs were remarkably consistent)
SPRING_FLOW                 0:00       ${p.flowRate}
SPRING_FLOW                 24:00      ${p.flowRate}

[PATTERNS]
;;Name           Type       Multipliers

[CURVES]
;;Name           Type       X-Value    Y-Value
`;

  return model;
}

/**
 * Persian Qanat SWMM5 Model
 */
function generateQanatModel(p, complexity) {
  const numShafts = Math.floor(p.length / p.shaftSpacing);
  const segmentsBetweenShafts = complexity === 'detailed' ? 2 : 1;
  const totalNodes = numShafts * segmentsBetweenShafts;
  
  let model = `
;; ============================================
;; PERSIAN QANAT MODEL
;; ============================================
;; Historical Context:
;; - Invented ~1000 BCE in Persia (modern Iran)
;; - Over 50,000 qanats still exist in Iran today
;; - Some are 70+ km long
;; - Yazd, Iran has qanats operating for 3,000 years
;;
;; Engineering Genius:
;; - Gravity-fed (zero energy required)
;; - Underground (minimal evaporation loss)
;; - Sustainable yield (taps aquifer, doesn't deplete)
;; - Mother well (maadar chah) reaches water table
;; - Vertical shafts every 20-50m for construction/maintenance
;;
;; Modern Relevance:
;; - UNESCO World Heritage recognition
;; - Being studied for sustainable water in arid regions
;; - Zero-carbon water transport inspiration
;; ============================================

[JUNCTIONS]
;;Name           Elevation      MaxDepth       InitDepth      SurDepth       Aponded
;; Mother well (maadar chah) - deepest point, reaches aquifer
MOTHER_WELL      50.00          ${(p.tunnelDiameter + 1).toFixed(2)}          0              0              0
`;

  // Create shaft nodes
  for (let i = 1; i <= numShafts; i++) {
    const distanceFromSource = i * p.shaftSpacing;
    const elevation = 50 + (distanceFromSource * p.slope);
    const shaftName = `SHAFT_${i}`;
    model += `;; Vertical shaft ${i} at ${distanceFromSource}m from mother well\n`;
    model += `${shaftName.padEnd(16)} ${elevation.toFixed(2).padStart(10)}     ${(p.tunnelDiameter + 1).toFixed(2).padStart(10)}     0              0              0\n`;
  }

  model += `
[OUTFALLS]
;;Name           Elevation      Type       Stage Data       Gated    Route To
;; Surface outlet (mazhar) - where qanat emerges
MAZHAR           ${(50 + p.length * p.slope + 1).toFixed(2)}          FREE                        NO

[CONDUITS]
;;Name           From Node        To Node          Length     Roughness  InOffset   OutOffset  InitFlow   MaxFlow
;; Underground tunnel sections (qanat gallery)
`;

  // Create tunnel segments
  let prevNode = 'MOTHER_WELL';
  for (let i = 1; i <= numShafts; i++) {
    const currentNode = `SHAFT_${i}`;
    const linkName = `TUNNEL_${i}`;
    model += `${linkName.padEnd(16)} ${prevNode.padEnd(16)} ${currentNode.padEnd(16)} ${p.shaftSpacing.toFixed(0).padStart(10)}     ${p.roughness.toFixed(4)}     0          0          0          0\n`;
    prevNode = currentNode;
  }
  model += `OUTLET_TUNNEL    SHAFT_${numShafts}        MAZHAR           50         ${p.roughness.toFixed(4)}     0          0          0          0\n`;

  model += `
[XSECTIONS]
;;Link           Shape        Geom1            Geom2      Geom3      Geom4      Barrels    Culvert
;; Circular tunnel - hand-dug by muqanni (qanat diggers)
`;

  for (let i = 1; i <= numShafts; i++) {
    model += `TUNNEL_${i}        CIRCULAR     ${p.tunnelDiameter}            0          0          0          1\n`;
  }
  model += `OUTLET_TUNNEL    CIRCULAR     ${p.tunnelDiameter}            0          0          0          1\n`;

  model += `
[INFLOWS]
;;Node           Constituent      Time Series      Type     Mfactor  Sfactor  Baseline Pattern
;; Groundwater inflow at mother well - sustainable aquifer yield
MOTHER_WELL      FLOW             AQUIFER_FLOW     FLOW     1.0      1.0      ${p.flowRate}

[TIMESERIES]
;;Name           Date       Time       Value
;; Qanat flow varies seasonally but is remarkably stable
AQUIFER_FLOW                0:00       ${p.flowRate}
AQUIFER_FLOW                6:00       ${(p.flowRate * 1.1).toFixed(4)}
AQUIFER_FLOW                12:00      ${p.flowRate}
AQUIFER_FLOW                18:00      ${(p.flowRate * 0.95).toFixed(4)}
AQUIFER_FLOW                24:00      ${p.flowRate}
`;

  return model;
}

/**
 * Trigger file download
 */
function exportSWMM5Model(inventionId) {
  const complexityEl = document.querySelector('input[name="model-complexity"]:checked');
  const complexity = complexityEl ? complexityEl.value : 'simple';
  
  const content = generateSWMM5File(inventionId, complexity);
  
  if (!content) return;
  
  // Create blob and download
  const blob = new Blob([content], { type: 'text/plain' });
  const url = URL.createObjectURL(blob);
  
  const a = document.createElement('a');
  a.href = url;
  a.download = `ancient_${inventionId}_model.inp`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
  
  // Show success message
  showExportSuccess(inventionId);
}

function showExportSuccess(inventionId) {
  const model = SWMM5_MODELS[inventionId];
  const toast = document.createElement('div');
  toast.className = 'export-toast';
  toast.innerHTML = `
    <div class="toast-icon">✅</div>
    <div class="toast-content">
      <strong>SWMM5 Model Downloaded!</strong>
      <p>${model.name}</p>
      <small>Open in EPA SWMM5 or compatible software</small>
    </div>
  `;
  document.body.appendChild(toast);
  
  setTimeout(() => {
    toast.classList.add('fade-out');
    setTimeout(() => toast.remove(), 500);
  }, 4000);
}
