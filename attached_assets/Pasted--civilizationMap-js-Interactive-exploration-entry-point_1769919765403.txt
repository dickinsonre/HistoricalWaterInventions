function renderCivilizationSelector() {
  const container = document.getElementById('civilization-grid');

  civilizations.forEach(civ => {
    const card = document.createElement('div');
    card.className = 'civilization-card';
    card.style.borderColor = civ.color;

    card.innerHTML = `
      <div class="civ-header" style="background: ${civ.color}20">
        <h3>${civ.name}</h3>
        <span class="period">${civ.period}</span>
      </div>
      <div class="civ-body">
        <p class="region">📍 ${civ.region}</p>
        <p class="description">${civ.description}</p>
        <div class="civ-stats">
          <span class="invention-count">${civ.inventionCount} inventions</span>
          <span class="key-invention">Key: ${civ.keyInvention}</span>
        </div>
      </div>
      <a href="/civilization/${civ.id}" class="explore-btn">
        Explore ${civ.name} →
      </a>
    `;

    container.appendChild(card);
  });
}
