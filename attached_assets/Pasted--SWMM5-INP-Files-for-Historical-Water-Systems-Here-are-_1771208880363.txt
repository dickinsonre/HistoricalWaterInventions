### Step 3: Update the Copy-to-Clipboard Function

```javascript
async function copyINP(modelId) {
  try {
    const model = models.find(m => m.id === modelId);
    const response = await fetch(model.inpFile);
    const inpContent = await response.text();
    await navigator.clipboard.writeText(inpContent);
    showToast(`${model.name} .INP copied to clipboard`);
  } catch (err) {
    console.error('Failed to copy:', err);
    showToast('Failed to copy — try again', 'error');
  }
}
```
