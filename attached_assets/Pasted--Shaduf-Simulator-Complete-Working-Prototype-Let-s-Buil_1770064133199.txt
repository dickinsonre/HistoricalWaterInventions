// ShadufSimulator.jsx
// Interactive Ancient Egyptian Shaduf - Educational Physics Simulator

import React, { useState, useEffect, useCallback } from 'react';

const ShadufSimulator = () => {
  // ============ PHYSICAL PARAMETERS ============
  const [params, setParams] = useState({
    // Lever arm lengths (meters)
    shortArm: 1.0,      // L1: pivot to counterweight
    longArm: 4.0,       // L2: pivot to bucket
    
    // Masses (kg)
    counterweight: 40,   // Mud brick or stone
    bucketEmpty: 2,      // Clay or leather bucket
    bucketVolume: 20,    // Liters capacity
    
    // Water properties
    waterLevel: 2.0,     // Depth to water surface (m)
    liftHeight: 1.5,     // Height above ground to discharge (m)
    
    // Operation
    cyclesPerHour: 30,   // Typical agricultural operation
  });

  // ============ SIMULATION STATE ============
  const [simulation, setSimulation] = useState({
    isRunning: false,
    angle: 20,           // Current lever angle (degrees from horizontal)
    bucketFill: 0,       // Current water in bucket (liters)
    phase: 'lowering',   // lowering | filling | lifting | emptying
    totalLifted: 0,      // Cumulative liters lifted
    cycleCount: 0,
    elapsedTime: 0,      // Seconds
  });

  // ============ PHYSICS CALCULATIONS ============
  const physics = {
    // Torque balance: τ₁ = τ₂
    // m₁ × g × L₁ = m₂ × g × L₂
    // Counterweight needed = (bucket + water) × (L₂/L₁)
    
    calculateCounterweightNeeded: () => {
      const waterMass = simulation.bucketFill; // 1 liter = 1 kg
      const totalBucketMass = params.bucketEmpty + waterMass;
      const ratio = params.longArm / params.shortArm;
      return totalBucketMass * ratio;
    },
    
    // Net torque determines motion
    calculateNetTorque: () => {
      const waterMass = simulation.bucketFill;
      const bucketSideTorque = (params.bucketEmpty + waterMass) * 9.81 * params.longArm;
      const counterweightTorque = params.counterweight * 9.81 * params.shortArm;
      return counterweightTorque - bucketSideTorque; // Positive = bucket rises
    },
    
    // Flow rate (liters per hour)
    calculateFlowRate: () => {
      const litersPerCycle = params.bucketVolume * 0.85; // 85% fill efficiency
      return litersPerCycle * params.cyclesPerHour;
    },
    
    // Work done per cycle (Joules)
    calculateWorkPerCycle: () => {
      const waterMass = params.bucketVolume * 0.85; // kg (water density = 1)
      const totalLift = params.waterLevel + params.liftHeight;
      return waterMass * 9.81 * totalLift;
    },
    
    // Human power required (Watts)
    calculatePowerRequired: () => {
      const workPerCycle = physics.calculateWorkPerCycle();
      const cycleTime = 3600 / params.cyclesPerHour; // seconds
      return workPerCycle / cycleTime;
    },
    
    // Mechanical advantage
    calculateMechanicalAdvantage: () => {
      return params.longArm / params.shortArm;
    },
    
    // Efficiency (accounting for friction, spillage)
    calculateEfficiency: () => {
      return 0.65; // Typical for manual shaduf operation
    }
  };

  // ============ ANIMATION LOOP ============
  useEffect(() => {
    if (!simulation.isRunning) return;
    
    const interval = setInterval(() => {
      setSimulation(prev => {
        let newAngle = prev.angle;
        let newFill = prev.bucketFill;
        let newPhase = prev.phase;
        let newTotalLifted = prev.totalLifted;
        let newCycleCount = prev.cycleCount;
        
        switch (prev.phase) {
          case 'lowering':
            // Bucket descends into water
            newAngle = Math.max(prev.angle - 2, -35);
            if (newAngle <= -35) {
              newPhase = 'filling';
            }
            break;
            
          case 'filling':
            // Bucket fills with water
            newFill = Math.min(prev.bucketFill + 4, params.bucketVolume * 0.85);
            if (newFill >= params.bucketVolume * 0.85) {
              newPhase = 'lifting';
            }
            break;
            
          case 'lifting':
            // Counterweight helps lift bucket
            newAngle = Math.min(prev.angle + 1.5, 45);
            if (newAngle >= 45) {
              newPhase = 'emptying';
            }
            break;
            
          case 'emptying':
            // Water pours into irrigation channel
            newFill = Math.max(prev.bucketFill - 5, 0);
            if (newFill <= 0) {
              newTotalLifted = prev.totalLifted + params.bucketVolume * 0.85;
              newCycleCount = prev.cycleCount + 1;
              newPhase = 'lowering';
            }
            break;
        }
        
        return {
          ...prev,
          angle: newAngle,
          bucketFill: newFill,
          phase: newPhase,
          totalLifted: newTotalLifted,
          cycleCount: newCycleCount,
          elapsedTime: prev.elapsedTime + 0.05,
        };
      });
    }, 50);
    
    return () => clearInterval(interval);
  }, [simulation.isRunning, params]);

  // ============ RENDER ============
  return (
    <div style={styles.container}>
      {/* Header */}
      <div style={styles.header}>
        <h1 style={styles.title}>⚖️ Ancient Egyptian Shaduf Simulator</h1>
        <p style={styles.subtitle}>
          New Kingdom Period (~1550-1077 BCE) • Interactive Physics Model
        </p>
      </div>

      <div style={styles.mainGrid}>
        {/* Left: Visualization */}
        <div style={styles.vizPanel}>
          <ShadufVisualization 
            angle={simulation.angle}
            bucketFill={simulation.bucketFill}
            params={params}
            phase={simulation.phase}
          />
        </div>

        {/* Right: Controls & Metrics */}
        <div style={styles.controlPanel}>
          {/* Simulation Controls */}
          <div style={styles.card}>
            <h3 style={styles.cardTitle}>🎮 Simulation Control</h3>
            <div style={styles.buttonRow}>
              <button
                style={{
                  ...styles.button,
                  ...(simulation.isRunning ? styles.buttonStop : styles.buttonStart)
                }}
                onClick={() => setSimulation(s => ({ ...s, isRunning: !s.isRunning }))}
              >
                {simulation.isRunning ? '⏸ Pause' : '▶ Start'}
              </button>
              <button
                style={{ ...styles.button, ...styles.buttonReset }}
                onClick={() => setSimulation({
                  isRunning: false,
                  angle: 20,
                  bucketFill: 0,
                  phase: 'lowering',
                  totalLifted: 0,
                  cycleCount: 0,
                  elapsedTime: 0,
                })}
              >
                🔄 Reset
              </button>
            </div>
            
            <div style={styles.phaseIndicator}>
              <span style={styles.phaseLabel}>Current Phase:</span>
              <span style={{
                ...styles.phaseBadge,
                backgroundColor: phaseColors[simulation.phase]
              }}>
                {phaseLabels[simulation.phase]}
              </span>
            </div>
          </div>

          {/* Parameter Sliders */}
          <div style={styles.card}>
            <h3 style={styles.cardTitle}>⚙️ Physical Parameters</h3>
            
            <SliderControl
              label="Short Arm (L₁)"
              value={params.shortArm}
              min={0.5}
              max={2.0}
              step={0.1}
              unit="m"
              onChange={(v) => setParams(p => ({ ...p, shortArm: v }))}
            />
            
            <SliderControl
              label="Long Arm (L₂)"
              value={params.longArm}
              min={2.0}
              max={6.0}
              step={0.1}
              unit="m"
              onChange={(v) => setParams(p => ({ ...p, longArm: v }))}
            />
            
            <SliderControl
              label="Counterweight Mass"
              value={params.counterweight}
              min={10}

              step={1}
              unit="kg"
              onChange={(v) => setParams(p => ({ ...p, counterweight: v }))}
            />
            
            <SliderControl
              label="Bucket Volume"
              value={params.bucketVolume}
              min={5}
              max={40}
              step={1}
              unit="L"
              onChange={(v) => setParams(p => ({ ...p, bucketVolume: v }))}
            />
            
            <SliderControl
              label="Water Depth"
              value={params.waterLevel}
              min={0.5}
              max={4.0}
              step={0.1}
              unit="m"
              onChange={(v) => setParams(p => ({ ...p, waterLevel: v }))}
            />
          </div>

          {/* Real-time Metrics */}
          <div style={styles.card}>
            <h3 style={styles.cardTitle}>📊 Performance Metrics</h3>
            
            <MetricRow

          </div>
        </div>
      </div>

      {/* Physics Explanation Panel */}
      <div style={styles.physicsPanel}>
        <h3 style={styles.cardTitle}>📐 Physics Principles</h3>
        <div style={styles.physicsGrid}>
          <div style={styles.physicsCard}>
            <h4>⚖️ Lever Equilibrium</h4>

// ============ SUB-COMPONENTS ============

const ShadufVisualization = ({ angle, bucketFill, params, phase }) => {
  const pivotX = 200;
  const pivotY = 120;
  const scale = 30; // pixels per meter
  
  // Calculate arm endpoints based on angle
  const angleRad = (angle * Math.PI) / 180;
  
  // Short arm (counterweight side) - goes opposite direction
  const shortEndX = pivotX - Math.cos(angleRad) * params.shortArm * scale;
  const shortEndY = pivotY - Math.sin(angleRad) * params.shortArm * scale;
  
  // Long arm (bucket side)
  const longEndX = pivotX + Math.cos(angleRad) * params.longArm * scale;
  const longEndY = pivotY + Math.sin(angleRad) * params.longArm * scale;
  
  // Water level
  const waterY = 220;
  
  // Bucket fill percentage
  const fillPercent = bucketFill / (params.bucketVolume * 0.85);
  
  return (
    <svg width="400" height="300" style={styles.svg}>
      {/* Sky gradient */}
      <defs>
        <linearGradient id="skyGradient" x1="0%" y1="0%" x2="0%" y2="100%">
          <stop offset="0%" stopColor="#87CEEB" />
          <stop offset="100%" stopColor="#E0F4FF" />
        </linearGradient>
        <linearGradient id="waterGradient" x1="0%" y1="0%" x2="0%" y2="100%">
          <stop offset="0%" stopColor="#4A90D9" />
          <stop offset="100%" stopColor="#2C5AA0" />
        </linearGradient>
        <linearGradient id="groundGradient" x1="0%" y1="0%" x2="0%" y2="100%">
          <stop offset="0%" stopColor="#8B7355" />
          <stop offset="100%" stopColor="#6B5344" />
        </linearGradient>
      </defs>
      
      {/* Background */}
      <rect x="0" y="0" width="400" height="180" fill="url(#skyGradient)" />
      
      {/* Ground */}
      <rect x="0" y="180" width="400" height="40" fill="url(#groundGradient)" />
      
      {/* Water */}
      <rect x="0" y={waterY}" width="180" height="80" fill="url(#waterGradient)" />
      
      {/* Irrigation channel */}
      <rect x="280" y="175" width="120" height="15" fill="#8B7355" />
      <rect x="285" y="178" width="110" height="9" fill="#4A90D9" opacity="0.7" />
      
      {/* Support structure */}
      <polygon 
        points={`${pivotX-15},180 ${pivotX+15},180 ${pivotX},${pivotY+10}`}
        fill="#5D4E37"
        stroke="#3D2E17"
        strokeWidth="2"
      />
      
      {/* Lever arm */}
      <line
        x1={shortEndX}
        y1={shortEndY}
        x2={longEndX}
        y2={longEndY}
        stroke="#8B4513"
        strokeWidth="6"
        strokeLinecap="round"
      />
      
      {/* Pivot point */}
      <circle cx={pivotX} cy={pivotY} r="8" fill="#5D4E37" stroke="#3D2E17" strokeWidth="2" />
      
      {/* Counterweight */}
      <rect
        x={shortEndX - 15}
        y={shortEndY - 10}
        width="30"
        height="25"
        fill="#8B7355"
        stroke="#5D4E37"
        strokeWidth="2"
        rx="3"
      />
      <text x={shortEndX} y={shortEndY + 5} textAnchor="middle" fontSize="10" fill="#FFF">
        {params.counterweight}kg
      </text>
      
      {/* Rope */}
      <line
        x1={longEndX}
        y1={longEndY}
        x2={longEndX}
        y2={longEndY + 40}
        stroke="#D2B48C"
        strokeWidth="2"
        strokeDasharray="4,2"
      />
      
      {/* Bucket */}
      <g transform={`translate(${longEndX - 15}, ${longEndY + 40})`}>
        {/* Bucket body */}
        <path
          d="M 0,0 L 5,35 L 25,35 L 30,0 Z"
          fill="#CD853F"
          stroke="#8B4513"
          strokeWidth="2"
        />
        {/* Water in bucket */}
        <rect
          x="5"
          y={35 - fillPercent * 30}
          width="20"
          height={fillPercent * 30}
          fill="#4A90D9"
          opacity="0.8"
        />
        {/* Bucket rim */}
        <ellipse cx="15" cy="0" rx="15" ry="4" fill="#CD853F" stroke="#8B4513" strokeWidth="2" />
      </g>
      
      {/* Phase indicator */}
      <text x="350" y="25" textAnchor="end" fontSize="12" fill="#333">
        {phaseLabels[phase]}
      </text>
      
      {/* Water drops animation during emptying */}
      {phase === 'emptying' && (
        <>
          <circle cx={longEndX + 5} cy={longEndY + 85} r="3" fill="#4A90D9">
            <animate attributeName="cy" from={longEndY + 85} to="190" dur="0.5s" repeatCount="indefinite" />
            <animate attributeName="opacity" from="1" to="0" dur="0.5s" repeatCount="indefinite" />
          </circle>
          <circle cx={longEndX - 5} cy={longEndY + 85} r="2" fill="#4A90D9">
            <animate attributeName="cy" from={longEndY + 85} to="190" dur="0.4s" repeatCount="indefinite" />
            <animate attributeName="opacity" from="1" to="0" dur="0.4s" repeatCount="indefinite" />
          </circle>
        </>
      )}
      
      {/* Labels */}
      <text x="90" y="260" textAnchor="middle" fontSize="11" fill="#FFF" fontWeight="bold">
        River Nile
      </text>
      <text x="340" y="172" textAnchor="middle" fontSize="10" fill="#333">
        Irrigation Canal
      </text>
    </svg>
  );
};

const downloadFile = (content, filename, mimeType) => {
  const blob = new Blob([content], { type: mimeType });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
};
